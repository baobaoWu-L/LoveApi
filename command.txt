========================================================================
 Love Api 平台 —— 本地编译 / 启动 / 账号 / 数据库架构 命令速查
========================================================================
（所有命令默认在仓库根目录 D:\LoveApiPlatform 下执行）
项目使用 docker compose -f docker-compose.mysql.yml 管理后端 + 前端 + Redis + MySQL。
前端由独立的 nginx 容器（love-api-frontend）承载，/api 与 /v1 反代到后端。

------------------------------------------------------------------------
一、Go 后端编译
------------------------------------------------------------------------
cd D:\LoveApiPlatform
go build ./...                         # 编译校验后端所有包
go vet ./...                           # 静态检查（可选）

# 重新构建后端镜像（改动 Go 代码或内嵌前端 dist 时）
docker build -f Dockerfile -t love-api:latest .

# 仅编译可执行文件，不打包 Docker（可选）
go build -o love-api ./main.go

------------------------------------------------------------------------
二、前端容器构建 / 启动
------------------------------------------------------------------------
# 构建前端镜像（npm 包原名 loveapi-web，容器内用 bun install + rsbuild build）
docker build -f Dockerfile.frontend -t love-api-frontend .

# 一键启动/更新所有服务（backend + frontend + redis + mysql）
docker compose -f docker-compose.mysql.yml up -d

# 仅重建前端容器 / 重启指定服务
docker compose -f docker-compose.mysql.yml up -d frontend
docker compose -f docker-compose.mysql.yml restart backend

# 查看日志
docker compose -f docker-compose.mysql.yml logs -f backend
docker compose -f docker-compose.mysql.yml logs -f frontend

# 停止全部
docker compose -f docker-compose.mysql.yml down

七、本地前端开发（可选，不用 Docker）
------------------------------------------------------------------------

bun run typecheck  # tsc 类型检查
bun run i18n:sync  # 同步 i18n 文案

以后本地更新代码后，服务器执行

# 进入项目目录

git status --short
# 查看服务器是否有本地修改

git pull --ff-only origin LoveApi

# 确认当前代码版本
构建两个前端：
cd /usr/src/itm/LoveApi/web/default
# 进入 default 前端

bun install
# 安装前端依赖

bun run build
# 构建 default 前端

cd /usr/src/itm/LoveApi/web/classic
# 进入 classic 前端

bun install
# 安装前端依赖

bun run build
# 构建 classic 前端
构建和替换后端：
cd /usr/src/itm/LoveApi
# 返回项目根目录

go mod download
# 下载 Go 依赖

go build -o bin/love-api.new .
# 编译临时后端程序

test -x bin/love-api.new && echo backend-build-ok
# 检查编译结果

mv bin/love-api.new bin/love-api
# 替换正式后端程序

chmod +x bin/love-api
# 确保后端有执行权限

systemctl restart love-api
# 重启后端服务

systemctl status love-api --no-pager
# 查看服务状态
后端日志命令
journalctl -u love-api -n 100 --no-pager
# 查看最近 100 行后端日志
journalctl -u love-api -f
# 实时查看后端日志，按 Ctrl+C 退出
journalctl -u love-api --since today --no-pager
# 查看今天的全部后端日志
journalctl -u love-api -p err..alert --since today --no-pager
# 只查看今天的错误日志
systemctl cat love-api
# 查看当前 systemd 服务配置
ss -lntp | grep ':3000'
# 检查后端是否监听 3000 端口
curl -sS -o /dev/null -w '%{http_code}\n' http://127.0.0.1:3000
# 测试本机访问后端
